        function initWebSocket(url, role) {
            if (!url) return;
            try {
                if (wsClient) wsClient.close();
                wsClient = new WebSocket(url);
                wsClient.onopen = () => console.log(`WebSocket connected as ${role}`);
                wsClient.onmessage = (event) => {
                    try {
                        const data = JSON.parse(event.data);
                        if (data.action === 'cue' && data.filename) {
                            const target = loadedSongs.find(s => s.filename.toLowerCase() === data.filename.toLowerCase());
                            if (target) renderSongSheet(target);
                        }
                    } catch (e) { console.error(e); }
                };
            } catch (e) { console.error(e); }
        }

        function switchSidebarTab(tab) {
            activeTab = tab;
            elements.tabSongsBtn.className = `flex-1 py-1.5 text-xs font-bold rounded-md ${tab === 'songs' ? 'bg-amber-500 text-slate-950' : 'text-slate-400 hover:bg-slate-800'}`;
            elements.tabSetsBtn.className = `flex-1 py-1.5 text-xs font-bold rounded-md ${tab === 'sets' ? 'bg-indigo-500 text-slate-950' : 'text-slate-400 hover:bg-slate-800'}`;
            elements.activeTabLabel.textContent = tab === 'songs' ? "Showing All Songs" : "Showing All Sets";
            renderSidebarList(elements.songSearch.value.toLowerCase());
        }

        function updateTabCounts() {
            elements.tabSongsCount.textContent = `(${loadedSongs.length})`;
            elements.tabSetsCount.textContent = `(${loadedSets.length})`;
        }

        function renderSidebarList(query = '') {
            elements.songList.innerHTML = '';
            if (activeTab === 'songs') {
                loadedSongs.sort((a, b) => a.title.localeCompare(b.title));
                const filtered = loadedSongs.filter(song => {
                    const matchesQuery = song.title.toLowerCase().includes(query) || (song.author && song.author.toLowerCase().includes(query));
                    const matchesFolder = selectedFolderFilter === 'ALL' || (song.folderPath || 'MAIN') === selectedFolderFilter;
                    return matchesQuery && matchesFolder;
                });
                filtered.forEach(song => {
                    const li = document.createElement('li');
                    const isActive = activeSong && activeSong.id === song.id;
                    li.className = `p-3 rounded-xl cursor-pointer transition flex items-center justify-between border ${isActive ? 'bg-amber-500/10 border-amber-500/30 text-amber-400 font-bold' : 'hover:bg-slate-800/60 text-slate-300 border-transparent'}`;
                    li.innerHTML = `
    ${song.title}
    ${song.author || 'Unknown'} (${song.folderPath || 'MAIN'})
    ${song.key || '?'}`;
                    li.addEventListener('click', () => { activeSet = null; renderSongSheet(song); });
                    elements.songList.appendChild(li);
                });
            } else {
                loadedSets.sort((a, b) => a.title.localeCompare(b.title));
                loadedSets.filter(s => s.title.toLowerCase().includes(query)).forEach(setObj => {
                    const li = document.createElement('li');
                    const isActive = activeSet && activeSet.id === setObj.id;
                    li.className = `p-3 rounded-xl cursor-pointer transition flex items-center justify-between border ${isActive ? 'bg-indigo-500/10 border-indigo-500/30 text-indigo-400 font-bold' : 'hover:bg-slate-800/60 text-slate-300 border-transparent'}`;
                    li.innerHTML = `
    ${setObj.title}
    ${setObj.songTitles.length} songs
    `;
                    li.addEventListener('click', () => renderSetSheet(setObj));
                    elements.songList.appendChild(li);
                });
            }
            lucide.createIcons();
        }

        function setupEventListeners() {
            elements.tabSongsBtn.addEventListener('click', () => switchSidebarTab('songs'));
            elements.tabSetsBtn.addEventListener('click', () => switchSidebarTab('sets'));
            elements.songSearch.addEventListener('input', (e) => renderSidebarList(e.target.value.toLowerCase()));
            elements.floatingMainBtn.addEventListener('click', () => elements.floatingMenu.classList.toggle('hidden'));
            elements.closeFloatingMenu.addEventListener('click', () => elements.floatingMenu.classList.add('hidden'));

            elements.openSettingsBtn.addEventListener('click', () => {
                elements.settingWsRole.value = appSettings.wsRole;
                elements.settingWsUrl.value = appSettings.wsUrl;
                elements.settingHeadingScaleSlider.value = appSettings.headingScale;
                elements.settingHeadingScaleLabel.textContent = `${appSettings.headingScale}%`;
                elements.settingChordScaleSlider.value = appSettings.chordScale;
                elements.settingChordScaleLabel.textContent = `${appSettings.chordScale}%`;
                elements.settingsModal.classList.remove('hidden');
            });
            elements.closeSettingsModal.addEventListener('click', () => elements.settingsModal.classList.add('hidden'));

            elements.settingHeadingScaleSlider.addEventListener('input', (e) => {
                elements.settingHeadingScaleLabel.textContent = `${e.target.value}%`;
            });
            elements.settingChordScaleSlider.addEventListener('input', (e) => {
                elements.settingChordScaleLabel.textContent = `${e.target.value}%`;
            });

            elements.saveSettingsBtn.addEventListener('click', () => {
                elements.settingsModal.classList.add('hidden');
                appSettings.wsRole = elements.settingWsRole.value;
                appSettings.wsUrl = elements.settingWsUrl.value;
                appSettings.headingScale = parseInt(elements.settingHeadingScaleSlider.value, 10);
                appSettings.chordScale = parseInt(elements.settingChordScaleSlider.value, 10);

                document.documentElement.style.setProperty('--heading-font-scale', `${appSettings.headingScale}%`);
                document.documentElement.style.setProperty('--chord-font-scale', `${appSettings.chordScale}%`);

                initWebSocket(appSettings.wsUrl, appSettings.wsRole);
            });
        }
